\section{How to Use This Handbook (read this page first)}\label{sec:howto}

\begin{prof}[The exam (from the professor)]
Two design questions (30 marks, scaled to 15). Each gives a \textbf{processor architecture}, a \textbf{workload}, and a \textbf{situation where it is not working ideally}. You must understand the problem \emph{from the question itself} (the clues give $\sim$80\% of the answer), state \textbf{assumptions}, explain \textbf{why} the problem occurs, redesign \textbf{the whole architecture}, and justify \textbf{why your design beats the alternatives} for \emph{this} scenario.
\end{prof}

\subsection*{The 6-move routine: do this for every question, in this order}
\begin{enumerate}[label=\textbf{\arabic*.}]
\item \textbf{Split the question into parts.} Every separate workload, scenario or symptom is one part (P1, P2, \ldots). On your rough sheet draw the \textbf{breakdown grid} (example on \pg{sec:breakdown}): \emph{Part $|$ Givens $|$ Clue words $|$ Cards $|$ Static/Dynamic}.
\item \textbf{Underline clue words} (``shared counter'', ``if inside the loop'', ``streaming'', ``unequal tasks'', ``battery''\ldots). Look each one up in the \textbf{Clue Index} (\pg{sec:clueidx}). It gives the meaning and the card IDs with page numbers.
\item \textbf{For the ``not working ideally'' part,} look the symptom up in the \textbf{Symptom Index} (\pg{sec:symidx}). It gives the root cause, how to prove it with a number, and the cards that fix it.
\item \textbf{Open each card.} Every card has the same fields (legend below). Check that its \emph{Use when} matches your clues; copy its \emph{Beats} line as your ``why better'' argument; add its \emph{Needs} line (what else must change); fill the \emph{Write} sentence with numbers from the question.
\item \textbf{Check the cross-cutting pages}: static vs.\ dynamic (\pg{sec:staticdyn}), will the memory hierarchy work (\pg{sec:memworks}), benefits/demerits of parallelism (\pg{sec:paraproscons}), knock-on effects (\pg{sec:knockon}).
\item \textbf{Write the answer in the 6-heading layout} (\pg{sec:layout}), copying the shape of the model answers (\pg{sec:modelA}). Finish with the checklist (\pg{sec:checklist}).
\end{enumerate}

\subsection*{Card legend (every concept card uses these fields)}
\small
\begin{tabularx}{\textwidth}{L{2.1cm}X}
\toprule
\tblhead{\textbf{Field} & \textbf{What it gives you in the answer}}\\
\midrule
\textbf{Fixes} & The problem this concept solves (the root cause it attacks). Use it to explain \emph{why the problem occurs}. \\
\textbf{How} & The mechanism in 1--3 lines. Writing the mechanism is what makes an answer non-generic. \\
\textbf{Use when} & The scenario clues that make this the right choice. Quote the matching clue from the question. \\
\textbf{Beats} & Why it is better than the named alternatives \emph{in that scenario}. This is your ``why my solution is better''. \\
\textbf{Cost / weak} & Its demerits: area, power, clock, complexity, overhead. Always state one. \\
\textbf{Avoid when} & When it is the wrong choice (use it to reject alternatives). \\
\textbf{Needs} & What else must change for it to work: the ``adjust the whole architecture'' part. \\
\textbf{Numbers} & Values from the slides/labs you can quote as evidence. \\
\textbf{Write} & A ready sentence: replace the brackets with this question's numbers and clues. \\
Tag (top right) & \textbf{Static} (decided at compile/design time) or \textbf{Dynamic} (decided at run time). \\
\bottomrule
\end{tabularx}

\begin{key}[Three rules that decide the marks]
\textbf{(1) Diagnose before you fix}: name the mechanism and prove it with one number. \textbf{(2) Every claim points at this question}: quote its clue or its number, never a bare textbook sentence. \textbf{(3) Every choice is compared}: ``X, not Y, because in this scenario Y fails at\ldots''. A red dagger\bs{} marks material not stated in the slides (textbook/general knowledge).
\end{key}
